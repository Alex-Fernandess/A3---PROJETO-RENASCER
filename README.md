comandos git                   

git init 
git config --global user.name ''                     
git config --gloval user.email ''                     

git init       //inicial o git                                    
git add . //adiciona as alterações                            
git commit -m '' //faz o commit                                                  
git remote add <nome remoto 'url' //se conecta ao github                            
git push <nome remoto> <nome branch> //envia para o github                              
git branch <nome branch> //cria a branch apenas                              
git checkout <nome branch> //muda para branch                               
git branch //lista todas as branch locais                              
git branch -r //lista as branch remotas                      
git log //lista os commits                       
git pull <nome remoto> <nome branch> //atualiza os commits da branch                       
git clone <url> //cria uma copia completa
